 };

 const apiKey=geminiKey();if(!apiKey){const e=new Error('Falta GEMINI_API_KEY para Zuzu.');e.status=503;throw e;}
 const isSynthesis=mode==='SYNTHESIS';
 // El contrato completo sigue validándose LOCALMENTE; el sobre sencillo evita
 // el HTTP 400 del proveedor con el AST recursivo y múltiples estados.

 const url=`https://generativelanguage.googleapis.com/v1beta/models/${encodeURIComponent(model)}:generateContent`;
 let repairFeedback='';
 // Reparación solo de la estructura, con el MISMO modelo y el MISMO contrato.
 // Nunca convierte una salida inválida en una acción ejecutable.
 for(let attempt=1;attempt<=3;attempt++){
   const started=Date.now();
   const systemText=baseInstruction+(repairFeedback?`\n\nVALIDACIÓN ANTERIOR FALLIDA (corrige el JSON completo, no eludir validaciones): ${repairFeedback}`:'');
   const body={
     systemInstruction:{parts:[{text:systemText}]},
     contents:[{role:'user',parts:[{text:JSON.stringify(packet)}]}],
     generationConfig:{responseMimeType:'application/json',responseJsonSchema:schema,maxOutputTokens:isSynthesis?2800:3800}
   };
   let res,payload;
   try{({res,payload}=await geminiFetchJsonWithTimeout(url,body,apiKey,Number(process.env.CONTROLEVENT_ZUZU_INTERACTIONS_TIMEOUT_MS||45000),externalSignal));}
   catch(error){zuzuTracePush(flowTrace,stage,'KO',cleanGeminiError(error),{model,attempt,transport:'generateContent+responseJsonSchema'});throw error;}
   const usage=usageSmall(payload,model);
   if(!res.ok){const e=new Error(payload?.error?.message||`Gemini Structured Output HTTP ${res.status}`);e.status=res.status;e.details=payload;zuzuTracePush(flowTrace,stage,'KO',cleanGeminiError(e),{model,usage,attempt,transport:'generateContent+responseJsonSchema',elapsedMs:Date.now()-started});throw e;}
   const raw=geminiOutText(payload);
   zuzuTracePush(flowTrace,stage,'OK',`Structured Output nativo · generateContent + responseJsonSchema · ${Date.now()-started} ms (intento ${attempt}).`,{model,usage,attempt,transport:'generateContent+responseJsonSchema'});
   try{
     if(!raw)throw new Error('Gemini no devolvió salida JSON estructurada v6_1-exp.');
     let parsed;
     try{parsed=JSON.parse(raw);}catch(error){throw new Error(`Gemini incumplió Structured Output nativo: ${cleanGeminiError(error)}`);}
     if(typeof parsed?.contract_json!=='string')throw new Error('Gemini no entregó contract_json en el sobre estructurado.');
     try{parsed=JSON.parse(parsed.contract_json);}catch(error){throw new Error(`contract_json de Gemini no es JSON válido: ${cleanGeminiError(error)}`);}
     return v6ValidateContract(parsed);
   }catch(error){
     repairFeedback=cleanGeminiError(error);
     zuzuTracePush(flowTrace,stage,attempt<3?'WARN':'KO',`Contrato interno rechazado (intento ${attempt}/3): ${repairFeedback}`,{model,attempt});
     if(attempt===3)throw new Error(`Zuzu no ha podido generar un contrato válido tras 3 intentos: ${repairFeedback}`);
   }
 }
}
